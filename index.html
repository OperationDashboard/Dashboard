Serverless UPSERT Database Config

  1. Create ONE Google Sheet. Name Columns A to L: Date, Code, Name, Sales, Trans, MyKasih, Lorry, Bank1, Bank2, Locked, SubmitTime, BankTime.
  2. Go to Extensions > Apps Script, paste the code below, and Deploy as Web App (Execute as: Me, Access: Anyone).
  3. Save the URL above. All 3500 branches and AMs will automatically push/pull from this single database. No manual syncing required!
function normalizeDate(d) {
  try {
    if (!d) return "";
    let dObj = new Date(d);
    if (isNaN(dObj.getTime())) return String(d).trim();
    let m = dObj.getMonth() + 1; let day = dObj.getDate();
    return dObj.getFullYear() + "-" + (m<10?"0":"")+m + "-" + (day<10?"0":"")+day;
  } catch(e) { return String(d).trim(); }
}

function doGet(e) {
  let ss = SpreadsheetApp.getActiveSpreadsheet();
  let action = e.parameter.action;
  
  if (action === 'get_branches') {
    let bSheet = ss.getSheetByName('Branches');
    if (!bSheet) return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON);
    let bData = bSheet.getDataRange().getValues();
    let branches = [];
    for (let i = 1; i < bData.length; i++) {
        if(bData[i][0]) branches.push({code: parseInt(bData[i][0]), name: String(bData[i][1]||"").trim(), am: String(bData[i][2]||"").trim(), state: String(bData[i][3]||"").trim(), warehouse: String(bData[i][4]||"").trim()});
    }
    return ContentService.createTextOutput(JSON.stringify(branches)).setMimeType(ContentService.MimeType.JSON);
  }
  
  let sheet = ss.getSheets()[0]; // Data jualan WAJIB di Tab Pertama (Sheet1)
  let data = sheet.getDataRange().getValues();
  let reqDate = e.parameter.date;
  let reqMonth = e.parameter.month;
  let result = [];
  for (let i = 1; i < data.length; i++) {
    let dStr = normalizeDate(data[i][0]);
    let match = (!reqDate && !reqMonth) || (reqDate && dStr === reqDate) || (reqMonth && dStr.substring(0,7) === reqMonth);
    if (match) {
      result.push({
        date: dStr, code: parseInt(data[i][1]), name: data[i][2],
        sales: parseFloat(data[i][3]), transactions: parseInt(data[i][4]), mykasih: parseFloat(data[i][5]), lorry: parseFloat(data[i][6]),
        bank1: parseFloat(data[i][7]), bank2: parseFloat(data[i][8]), night_locked: data[i][9] == 1,
        night_submit_time: data[i][10], bank1_time: data[i][11], bank2_time: data[i][12]
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  let ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheets()[0]; // Data jualan WAJIB di Tab Pertama (Sheet1)
  let payload = JSON.parse(e.postData.contents);
  let action = payload.action; let rec = payload.data;
  let data = sheet.getDataRange().getValues();
  let rowIndex = -1;
  for(let i = 1; i < data.length; i++) {
    let dStr = normalizeDate(data[i][0]);
    if(rec && dStr === rec.date && data[i][1] == rec.code) { rowIndex = i + 1; break; }
  }
  
  if (action === 'batch_upsert') {
    let batchData = payload.data;
    if(!batchData || !batchData.length) return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
    
    // Process batch upsert
    for(let r of batchData) {
      let rIdx = -1;
      for(let i = 1; i < data.length; i++) {
        let dStr = normalizeDate(data[i][0]);
        if(dStr === r.date && data[i][1] == r.code) { rIdx = i + 1; break; }
      }
      let rowData = [r.date, r.code, r.name, r.sales||0, r.transactions||0, r.mykasih||0, r.lorry||0, r.bank1||0, r.bank2||0, r.night_locked?1:0, r.night_submit_time?"'"+r.night_submit_time:"", r.bank1_time?"'"+r.bank1_time:"", r.bank2_time?"'"+r.bank2_time:""];
      if (rIdx > -1) {
        sheet.getRange(rIdx, 1, 1, 13).setValues([rowData]);
        data[rIdx-1] = rowData; // Update local data array so next iterations find it
      } else {
        sheet.appendRow(rowData);
        data.push(rowData); // Update local data array
      }
    }
  } else if (action === 'upsert') {
    let rowData = [rec.date, rec.code, rec.name, rec.sales||0, rec.transactions||0, rec.mykasih||0, rec.lorry||0, rec.bank1||0, rec.bank2||0, rec.night_locked?1:0, rec.night_submit_time?"'"+rec.night_submit_time:"", rec.bank1_time?"'"+rec.bank1_time:"", rec.bank2_time?"'"+rec.bank2_time:""];
    if (rowIndex > -1) sheet.getRange(rowIndex, 1, 1, 13).setValues([rowData]);
    else sheet.appendRow(rowData);
  } else if (action === 'reset_night' && rowIndex > -1) {
    sheet.getRange(rowIndex, 4, 1, 4).setValues([[0, 0, 0, 0]]);
    sheet.getRange(rowIndex, 10, 1, 2).setValues([[0, ""]]);
  } else if (action === 'reset_bank' && rowIndex > -1) {
    sheet.getRange(rowIndex, 8, 1, 2).setValues([[0, 0]]);
    sheet.getRange(rowIndex, 12, 1, 2).setValues([["", ""]]);
  } else if (action === 'master_reset') {
    for(let i = data.length - 1; i >= 1; i--) {
      let dStr = normalizeDate(data[i][0]);
      if(dStr === rec.date) {
         sheet.getRange(i+1, 4, 1, 4).setValues([[0, 0, 0, 0]]);
         sheet.getRange(i+1, 10, 1, 2).setValues([[0, ""]]);
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
